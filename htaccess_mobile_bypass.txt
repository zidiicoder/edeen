# Add this to the TOP of your public/.htaccess file on the server
# This allows all mobile app requests to bypass IP restrictions

<IfModule mod_rewrite.c>
    # Allow mobile app user agents
    RewriteCond %{HTTP_USER_AGENT} "Android|iPhone|iPad|Mobile" [NC]
    RewriteRule .* - [E=MOBILE:1]
</IfModule>

# Disable ModSecurity for mobile app requests
<IfModule mod_security.c>
    SecRuleEngine Off
</IfModule>
